\frame{
\frametitle{Reduction: $\beta$ rules}

\hfill\fbox{$\Delta \vdash M \reduce N$}

\vspace{20pt}
\begin{align*}
  \Delta \vdash (\lam{x\of A}N)\,W
    & \reduce N[x\by W] \\[2ex]
  \Delta \vdash \tyapp{(\Lam{X}V_B)}{A}
    &\reduce  \boundary{V}{\bind{X}{A}}{\unsealat{X}{B}{A}}
\end{align*}


}
%===============================================================================
\frame{
\frametitle{Reduction: elimination crosses a boundary}

\hfill\fbox{$\Delta \vdash M \reduce N$}

\begin{align*}
 \Delta \vdash \left(\boundary{V}{\Theta}{s \mapsto t}\right)\,W
    &\reduce \boundary{V\,\left(\boundary{W}{\dual{\Theta}}{s}\right)}{\Theta}{t} \\[2ex]
 \Delta \vdash
         \tyapp{\left(\boundary{V}{\Theta}{\forall X.\,s}\right)}{A}
       &\reduce \boundary{\tyapp{V}{X}}{\bind{X}{A},\Theta}{s}
\end{align*}

}
%===============================================================================
\frame{
\frametitle{Reduction: cancellation and identity}

\hfill\fbox{$\Delta \vdash M \reduce N$}
{\small
\begin{align*}
 \Delta \vdash \boundary{\boundary{V}{\Theta_1}{\seal{X}}}{\Theta_2}{\unseal{X}}
       &\reduce \boundary{V}{\owners{\Theta_2}}{\idc{A}} \\[2ex]
       \quad\text{where }\fce{\Theta_2}{\Delta} \ni X := A \\[2ex]
 \Delta \vdash \boundary{k}{\Theta}{\kw{id}\,\iota}
       &\reduce k \\[2ex]
 \Delta \vdash \boundary{\boundary{V}{\Theta_1}{\kw{id}\,X}}
                  {\Theta_2}{\unseal{X}}
       &\reduce \boundary{\boundary{V}{\Theta_1 + \Theta_2}{\unseal{X}}}
                 {\emptyset}{\idc{A}} \\[2ex]
       \quad\text{where }\fce{\Theta_2}{\Delta} \ni X := A
\end{align*}
}
\vfill
{\small $\owners{\Theta}$ retains the $X{:=}A$ entries of
$\Theta$ and drops its $\kw{lock}$ and $\kw{unlock}$ entries.}
}
%===============================================================================
\frame{
\frametitle{Reduction: congruence}

\hfill\fbox{$\Delta \vdash M \reduce N$}

\begin{gather*}
  \inference
    {\Delta \vdash L \reduce L'}
    {\Delta \vdash L\,M \reduce L'\,M}
  \quad
  \inference
    {\Delta \vdash M \reduce M'}
    {\Delta \vdash V\,M \reduce V\,M'}
  \\[2ex]
  \inference
    {\Delta \vdash L \reduce L'}
    {\Delta \vdash \tyapp{L}{A}
      \reduce \tyapp{L'}{A}}
  \quad      
  \inference
    {\Delta,X \vdash N \reduce N'}
    {\Delta \vdash \Lam{X}N \reduce \Lam{X}N'}
  \\[2ex]
  \inference
    {\intc{\Theta}{\Delta} \vdash M \reduce M'}
    {\Delta \vdash
       \boundary{M}{\Theta}{c}
       \reduce
       \boundary{M'}{\Theta}{c}}
\end{gather*}
}

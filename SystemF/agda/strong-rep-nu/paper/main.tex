Reduction
 \hfill \fbox{$\Xi \vdash M \becomes N \dashv \xi $}
\begin{align}
  \Xi \vdash \op(\vec{V})
    & \becomes~ \sem{\op}(\vec{V}) \dashv \epsilon
    \label{Delta}\tagsc{Delta}  \\
  \Xi \vdash (\lam{x \of A} N) \app V
    & \becomes~ N[x \by V] \dashv \epsilon
    \label{Beta}\tagsc{Beta}  \\
  \Xi \vdash \boundry{U}{\delta}{c \tu d} \app W &
    \becomes \boundry{U \app (\boundry{W}{\dual{\delta}}{c})}{\delta}{d}
    \dashv \epsilon 
    \label{Wrap}\tagsc{Wrap} \\
  \Xi \vdash \nuinst{X}{A}{(\Lam{X}V)}{d} &\becomes~
          \boundry{V}{{+X}^\alpha}{d} \dashv \alpha\by \Xi(A)
          \label{TyBeta}\tagsc{TyBeta}\\
   \Xi \vdash \nuinst{X}{A}{\boundry{(\Lam{X}V)}{\delta}{\all{X}c}}{d} &
       \becomes~ \boundry{(\boundry{V}{\delta}{c})}{{+X}^\alpha}{d}
        \dashv \alpha\by \Xi(A)
        \label{TyWrap}\tagsc{TyWrap} \\
   \Xi \vdash  \boundry{(\boundry{U}{\delta_1}{t_1})}{\delta_2}{c_2} &
     \becomes~ \boundry{U}{\delta_2 \mathop{++} \delta_1}{\Xi' \vdash t_1 \fatsemi c_2}  \dashv \epsilon \label{Merge}\tagsc{Merge} \\
     & \text{where } \Xi' = (\delta_2 \mathop{++} \delta_1)(\Xi) \notag \\
   \Xi \vdash \boundry{U}{\delta}{\id{K}} &\becomes~ U \dashv \epsilon
     \label{Id}\tagsc{Id}
\end{align}

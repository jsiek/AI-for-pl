\frame{
\frametitle{Syntactic Type Abstraction (Grossman et al.)}

\begin{itemize}
\item Define a multi-agent calculus that supports the \emph{embedding}
  of an agent within another agent.

\item The agents have differing knowledge about type equalities
  that link abstract types to their representations.

\item System F is modeled by the body of each $\Lambda$ spawning a new
  agent. Type application gives the knowledge (e.g. $X = A$) to the
  surrounding agent, not the spawned one.

\item The typing of embedding says the inner and outer types must
  match using the knowledge of the agents involved.

\item Their type system does not include well-formedness conditions to
  check type variable scoping.

\item Type variables are added but never removed from the typing context.
\end{itemize}

}

\frame{
  \frametitle{An example that breaks Fitch-style locking}
  \small
  The term $\rev{M}{\bind{X}{A}}$ reveals $X$ outward,
  $\cnc{M}{\bind{X}{A}}$ conceals it; the interior of a conceal was typed at
  $\Gamma\!\downarrow\!X$, the context with $X$ \emph{and everything bound
  after it} dropped.
  \begin{align*}
    &\app{\tyapp{\big(\app{\tyapp{(\rX{\Lam{X} \lam{f\of(\forall Z.\,Z\func Z)} \Lam{Y}
  \tyapp{f}{Y}})}{\Nat}}{(\rZ{\idZ})}\big)}{\Bool}}{\kw{true}}\\
    \reduce\;&\app{\tyapp{\big(\app{\rev{(\rX{\lam{f\of(\forall Z.\,Z\func Z)} \Lam{Y}
  \tyapp{f}{Y}})}{\bind{X}{\Nat}}}{(\rZ{\idZ})}\big)}{\Bool}}{\kw{true}}
       \\
    \reduce\;&\app{\tyapp{\big(\rev{(\rX{\app{(\lam{f}\Lam{Y}\tyapp{f}{Y})}{\cnc{(\rZ{\idZ})}{\bind{X}{\Nat}}}})}{\bind{X}{\Nat}}
  \big)}{\Bool}}{\kw{true}}
       \\
    \reduce\;&\app{\tyapp{\big(\rev{(\rX{\Lam{Y}
  \tyapp{\cnc{(\rZ{\idZ})}{\bind{X}{\Nat}}}{Y}})}{\bind{X}{\Nat}}\big)}{\Bool}}{\kw{true}}
       \\
    \reduce\;&\app{\tyapp{\big(\rev{(\rX{\Lam{Y}
  \cnc{(\tyapp{(\rZ{\idZ})}{Y})}{\bind{X}{\Nat}}})}{\bind{X}{\Nat}}\big)}{\Bool}}{\kw{true}}
  \end{align*}

  The last term is ill-typed because $\tyapp{(\idZ)}{Y}$ refers to $Y$
  which is blocked by the $\downarrow \bind{X}{\Nat}$.

  
  %% The rule that fired:
  %% \[
  %%   \tyapp{\cnc{F}{\bind{X}{A}}}{C} \;\reduce\; \cnc{(\tyapp{F}{C[X{:=}A]})}{\bind{X}{A}}
  %% \]
  %% It pushes the type argument $Y$ \emph{into} the concealed body, whose
  %% context is $(Y,\,X{:=}\Nat)\!\downarrow\!X = \emptyset$: the conceal was born
  %% before $\Lambda Y$ and drifted under it.  $\tyapp{(\idZ)}{Y}$ names a
  %% variable its context does not have.
}

\begingroup\small
\[

\frame{
  \frametitle{Operations on type contexts}

{\fontsize{8}{9}\selectfont
\hfill\fbox{$\mask{X}{\Delta} = \Delta'$} \\
\[
\begin{aligned}
\mask{X}{\emptyset} &= \emptyset \\
\mask{X}{(\Delta,\beta)} &=
\begin{cases}
\Delta,\maskentry{\beta} &\text{if }\entryname{\beta}=X \\
\mask{X}{\Delta},\beta &\text{otherwise}
\end{cases} \\
\mask{X}{(\Delta,\maskentry{\beta})} &= \mask{X}{\Delta},\maskentry{\beta} 
\end{aligned}
\]
\hfill\fbox{$\unmask{X}{\Delta} = \Delta'$}
\[
\begin{aligned}
\unmask{X}{\emptyset} &= \emptyset \\
\unmask{X}{(\Delta,\maskentry{\beta})} &= \Delta,\beta
&&\text{if }\entryname{\beta}=X \\
\unmask{X}{(\Delta,\hat{\beta})} &= \unmask{X}{\Delta},\hat{\beta}
&&\text{if }X\neq\entryname{\hat{\beta}}
\end{aligned}
\]

\begin{columns}[T]
\begin{column}{0.5\textwidth}
  \hfill\fbox{$\delta(\Delta) = \Delta'$}
  \[
\begin{aligned}
\emptyset(\Delta) &= \Delta \\
(\lockX{X},\delta)(\Delta) &= \mask{X}{\delta(\Delta)} \\
(\unlockX{X},\delta)(\Delta) &= \unmask{X}{\delta(\Delta)}
\end{aligned}
\]
\end{column}
\begin{column}{0.5\textwidth}
  \hfill\fbox{$\delta^{+}(\Delta) = \Delta'$}
\[
\begin{aligned}
\emptyset^{+}(\Delta) &= \Delta \\
(\lockX{X},\delta)^{+}(\Delta) &= \delta^{+}(\Delta) \\
(\unlockX{X},\delta)^{+}(\Delta) &= \unmask{X}{\delta^{+}(\Delta)}
\end{aligned}
\]
\end{column}
\end{columns}
}

\hfill\fbox{$\Delta \vdash M \reduce N$}

\begin{align*}

\frame{
\frametitle{Preservation}

\begin{center}
A reduction step preserves the type of a closed term.
\end{center}

\vspace{24pt}


\[
\begin{aligned}
&\text{If }\Delta\mid\emptyset\vdash M:A
  \text{ and }\Delta\vdash M\reduce M' \\
&\text{then } \Delta\mid\emptyset\vdash M':A.
\end{aligned}
\]

}

\frame{
\frametitle{Determinism}

\begin{center}
Every term has at most one immediate reduct.
\end{center}

\vspace{24pt}

\[
\begin{aligned}
&\text{If }\Delta\vdash M\reduce M_1
  \text{ and }\Delta\vdash M\reduce M_2 \\
&\text{then } M_1=M_2.
\end{aligned}
\]
